# 10. 新增模块模板

## 1. 头文件模板

```c
#ifndef PROJECT_MODULE_H
#define PROJECT_MODULE_H

#include <stdbool.h>
#include <stdint.h>

typedef enum
{
    MODULE_STATUS_OK = 0,
    MODULE_STATUS_ERROR,
    MODULE_STATUS_TIMEOUT
} module_status_t;

void module_init(void);
module_status_t module_process(void);

#endif /* PROJECT_MODULE_H */
```

## 2. C 文件模板

```c
#include "module.h"

#include "stm32f4xx.h"

static volatile bool module_event_pending;

void module_init(void)
{
    /* 1. 使能时钟；2. 配置 GPIO；3. 配置外设；4. 配置中断； */
    module_event_pending = false;
}

module_status_t module_process(void)
{
    if (!module_event_pending)
    {
        return MODULE_STATUS_OK;
    }

    module_event_pending = false;
    return MODULE_STATUS_OK;
}
```
